Run TACO on a split

pip install -e .
export TACO_API_KEY=… TACO_MODEL=gpt-4o-mini
taco run --input benchmark/TACO-Judge/chatgpt4o.jsonl \
         --output outputs/chatgpt4o.jsonl \
         --question-field question --answer-field llmanswer \
         --language python --resume
